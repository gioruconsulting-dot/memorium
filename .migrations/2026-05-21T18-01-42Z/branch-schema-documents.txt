CREATE TABLE documents (                                                                                                                                             
    id TEXT PRIMARY KEY,                                                                                                                                             
    user_id TEXT NOT NULL REFERENCES users(id) ON DELETE CASCADE,                                                                                                    
    title TEXT NOT NULL,                                                                                                                                             
    content TEXT NOT NULL,                                                                                                                                           
    themes TEXT,                                                                                                                                                     
    question_count INTEGER NOT NULL,                                                                                                                                 
    created_at INTEGER NOT NULL                                                                                                                                      
  , is_public INTEGER NOT NULL DEFAULT 1, description TEXT, topic TEXT, concepts_json TEXT, source_type TEXT NOT NULL DEFAULT 'uploaded'                             
  CHECK (source_type IN ('uploaded', 'note')), note_draft_content TEXT, last_generated_at INTEGER, updated_at INTEGER, note_version INTEGER NOT NULL DEFAULT 0);
